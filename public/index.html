Behind the Scenes: The Cash-Flow Engine

A collegiate deep dive into our Java graph debt minimization algorithm.

DebtSimplifier.java Java 25 LTS 
public static List<Transaction> simplifyDebts(List<Transaction> initialDebts) {
    // 1. Compute Net Balances for each member
    Map<String, Double> netBalances = new HashMap<>();
    for (Transaction t : initialDebts) {
        netBalances.merge(t.from, -t.amount, Double::sum);
        netBalances.merge(t.to, t.amount, Double::sum);
    }

    // 2. Dual Priority Queues for Greedy Matching
    PriorityQueue<ParticipantBalance> debtors = new PriorityQueue<>();
    PriorityQueue<ParticipantBalance> creditors = new PriorityQueue<>();

    // 3. Match Max Debtor with Max Creditor (O(V log V))
    while (!debtors.isEmpty() && !creditors.isEmpty()) {
        double settle = Math.min(-debtor.netAmount, creditor.netAmount);
        transactions.add(new Transaction(debtor.name, creditor.name, settle));
        // update balances and re-queue
    }
    return transactions;
}
